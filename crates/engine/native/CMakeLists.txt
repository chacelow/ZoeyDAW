    $<TARGET_OBJECTS:zoeydaw_playback_runtime>)

target_compile_definitions(zoeydaw_playback PUBLIC
    JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0
    JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_PLUGINHOST_AU=0 JUCE_PLUGINHOST_VST3=0
    JUCE_PLUGINHOST_LADSPA=0 JUCE_ASIO=0 JUCE_WASAPI=1
    TRACKTION_ENABLE_TIMESTRETCH_SOUNDTOUCH=1 TRACKTION_ENABLE_TIMESTRETCH_SIGNALSMITH=1
    TRACKTION_ENABLE_TIMESTRETCH_RUBBERBAND=0 TRACKTION_ENABLE_TIMESTRETCH_ELASTIQUE=0 TRACKTION_LOG_DEVICES=0)
# JUCE module targets are INTERFACE targets: their source files become members of this archive.
target_link_libraries(zoeydaw_playback PRIVATE tracktion::tracktion_engine
    tracktion::tracktion_graph tracktion::tracktion_core juce::juce_cryptography juce::juce_dsp)
if(MSVC)
    target_compile_options(zoeydaw_playback PRIVATE /EHsc)
endif()
add_library(zoeydaw_native_compile_config INTERFACE)
target_include_directories(zoeydaw_native_compile_config INTERFACE
    $<TARGET_PROPERTY:zoeydaw_playback,INCLUDE_DIRECTORIES>)
target_compile_definitions(zoeydaw_native_compile_config INTERFACE
    $<TARGET_PROPERTY:zoeydaw_playback,COMPILE_DEFINITIONS>)
target_compile_options(zoeydaw_native_compile_config INTERFACE
    $<TARGET_PROPERTY:zoeydaw_playback,COMPILE_OPTIONS>)
foreach(module model analysis runtime)
    target_link_libraries(zoeydaw_playback_${module} PRIVATE zoeydaw_native_compile_config)
    target_include_directories(zoeydaw_playback_${module} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/${module}" "${CMAKE_CURRENT_SOURCE_DIR}/model")
endforeach()
